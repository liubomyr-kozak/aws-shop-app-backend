# (A) Спробуй знести всі стеки CDK
cdk destroy --all --force --profile cloudx-js --region us-east-2

# (B) Якщо стек DELETE_FAILED, знайди бакети і очисти (підстав свій стек)
aws cloudformation list-stack-resources \
--stack-name DeployWebAppStack \
--region us-east-2 --profile cloudx-js \
--query 'StackResourceSummaries[?ResourceType==`AWS::S3::Bucket`].PhysicalResourceId' \
--output text

# (припустимо, бакет називається my-webapp-bucket-123)
aws s3 rm s3://my-webapp-bucket-123 --recursive --region us-east-2 --profile cloudx-js
# (за потреби — версії+delete-markers, див. скрипт вище)

# (C) Видали стек і дочекайся
aws cloudformation delete-stack --stack-name DeployWebAppStack --region us-east-2 --profile cloudx-js
aws cloudformation wait stack-delete-complete --stack-name DeployWebAppStack --region us-east-2 --profile cloudx-js

# (D) Прибери CDKToolkit (bootstrap) — якщо хочеш чистити «під нуль»
aws cloudformation delete-stack --stack-name CDKToolkit --region us-east-2 --profile cloudx-js
aws cloudformation wait stack-delete-complete --stack-name CDKToolkit --region us-east-2 --profile cloudx-js

# (E) Видали можливі ручні Lambda/LogGroup
aws lambda delete-function --function-name getProductsList  --region us-east-2 --profile cloudx-js || true
aws logs delete-log-group --log-group-name /aws/lambda/getProductsList --region us-east-2 --profile cloudx-js || true
